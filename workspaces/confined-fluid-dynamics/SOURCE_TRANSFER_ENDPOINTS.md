### Method B — exact binary bytes, one blob and one guarded commit
```js
const repo = "BestNameYet/IssueStack-";

const headRef = await tools.mcp__GitHub__fetch({url: base + "/git/ref/heads/main"});

const priorTree = JSON.parse(prior.result.content).tree.sha;
const blob = await tools.mcp__GitHub__create_blob({
  repository_full_name: repo,
  content: literalBase64Bytes, // entire staged file, base64 encoded (not a path)
  encoding: "base64"
});
const tree = await tools.mcp__GitHub__create_tree({
  repository_full_name: repo, base_tree_sha: priorTree,
  tree_elements: [{path: repositoryPath, mode: "100644", type: "blob", sha: blob.result.sha}]
});
const commit = await tools.mcp__GitHub__create_commit({
  repository_full_name: repo, message: "Import binary source: " + repositoryPath,
  tree_sha: tree.result.sha, parent_sha: parent
});
const pushed = await tools.mcp__GitHub__update_ref({
  repository_full_name: repo, branch_name: "main", sha: commit.result.sha,
  force: false, expected_sha: parent
});
if (!pushed.result.success) throw Error("Branch ref update not verified");
// Re-fetch the branch/file and verify the committed Git blob ID against local bytes.
```
`create_tree` uses an existing base tree; do not omit `base_tree_sha`. Recheck the latest branch head for every write and stop on conflict. For bulk transfer, batch multiple entries into one tree/commit and perform one guarded ref update, with every blob independently verified. Note that `tools` in the connected runner cannot automatically access `/mnt/data`; the base64/text must be transported to it by an actual supported interface.

## 3. Extract the two ZIP collections into `/mnt/data`

This Python is a local **staging** operation only, not a GitHub upload. It retains filename characters and verifies archive-relative paths against traversal. Staging paths match the table below.
```python
from pathlib import Path, PurePosixPath
from zipfile import ZipFile
from io import BytesIO

DATA = Path("/mnt/data")
OUT = DATA / "cfd_source_staging"
OUT.mkdir(parents=True, exist_ok=True)

def write_member(data, dst):
    dst = Path(dst)
    assert dst.resolve().is_relative_to(OUT.resolve()), "unsafe extraction path"
    dst.parent.mkdir(parents=True, exist_ok=True)
    dst.write_bytes(data)

outer = DATA / "Confined_Fluid_Dynamics_Renamed_Source_Items(1).zip"
prefix = PurePosixPath("Confined_Fluid_Dynamics_Renamed_Source_Items")
with ZipFile(outer) as z:
    for i in z.infolist():
        if i.is_dir(): continue
        rel = PurePosixPath(i.filename).relative_to(prefix)
        assert not rel.is_absolute() and ".." not in rel.parts
        write_member(z.read(i), OUT / "renamed" / str(rel))
    for archive, group in [
        ("Antiparallel_Vortex_Tubes_1988_and_2023_Paper_Pair.zip", "antiparallel"),
        ("Fluidic_Oscillators_Impinging_Jets_and_Cavitation_Reference_Set.zip", "fluidic")]:
        nested_bytes = z.read(str(prefix / archive))
        with ZipFile(BytesIO(nested_bytes)) as sub:
            for i in sub.infolist():
                if i.is_dir(): continue
                rel = PurePosixPath(i.filename)
                assert not rel.is_absolute() and ".." not in rel.parts
                write_member(sub.read(i), OUT / "nested" / group / str(rel))

with ZipFile(DATA / "ASO-Standard-v1.0.0.zip") as z:
    prefix = PurePosixPath("ASO-Standard-v1.0.0")
    for i in z.infolist():
        if i.is_dir(): continue
        rel = PurePosixPath(i.filename).relative_to(prefix)
        assert not rel.is_absolute() and ".." not in rel.parts
        write_member(z.read(i), OUT / "aso" / str(rel))
```
